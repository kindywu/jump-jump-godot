# Jump Jump Godot — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 1:1 port of the Bevy jump-jump game to Godot 4.6 using GDScript

**Architecture:** Autoload `GameState` manages state machine and score. Root scene `main.tscn` contains camera, light, ground, and dynamically-managed platform/player instances. Game logic lives in node-attached scripts (`player.gd`, `platform.gd`, `camera.gd`). UI is rendered on a `CanvasLayer` with Control nodes. Math-based collision detection (no physics engine). Animations via manual `_process` delta calculations matching the Bevy math exactly.

**Tech Stack:** Godot 4.6, GDScript, Jolt Physics (not used for gameplay), GPUParticles3D

extends Node

enum State { MAIN_MENU, PLAYING, GAME_OVER }

signal state_changed(new_state: State)
signal score_changed(new_score: int)
signal score_up(landing_pos: Vector3)

var current_state: State = State.MAIN_MENU
var score: int = 0

func change_state(new_state: State) -> void:
	current_state = new_state
	state_changed.emit(new_state)

func add_score() -> void:
	score += 1
	score_changed.emit(score)

func reset_score() -> void:
	score = 0
	score_changed.emit(0)

- [ ] **Step 1: Write player.gd**

Create `res://scripts/player.gd`:
```gdscript
extends Node3D

const INITIAL_POS := Vector3(0.0, 1.5, 0.0)
const JUMP_SPEED := 3.0
const MAX_SCALE_XZ := 1.3
const MIN_SCALE_Y := 0.6
const CHARGE_XZ_RATE := 0.12
const CHARGE_Y_RATE := 0.15
const FALL_SPEED := 0.7

enum FallType { STRAIGHT, TILT }

var charging := false
var charge_start_time := 0.0

# Jump
var jump_active := false
var jump_start_pos := Vector3.ZERO
var jump_end_pos := Vector3.ZERO
var jump_duration := 0.0
var jump_falled := false

# Fall
var fall_active := false
var fall_type: FallType = FallType.STRAIGHT
var fall_tilt_dir := Vector3.ZERO
var fall_pos := Vector3.ZERO
var fall_tilt_done := false
var fall_played_sound := false

# Audio refs
@onready var accumulation_player: AudioStreamPlayer3D = $AccumulationSound
@onready var success_player: AudioStreamPlayer3D = $SuccessSound
@onready var fall_player: AudioStreamPlayer3D = $FallSound

# Particle ref
@onready var charge_particles: GPUParticles3D = $ChargeParticles
@onready var charge_particle_timer: Timer = $ChargeParticleTimer

func _ready() -> void:
	transform.origin = INITIAL_POS

func _process(delta: float) -> void:
	if GameState.current_state != GameState.State.PLAYING:
		return

	if charging:
		_animate_charge(delta)
	if jump_active:
		_animate_jump(delta)
	if fall_active and not jump_active:
		_animate_fall(delta)

func _unhandled_input(event: InputEvent) -> void:
	if GameState.current_state != GameState.State.PLAYING:
		return
	
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
		if event.pressed and not jump_active and not fall_active:
			_start_charge()
		elif not event.pressed and charging:
			_do_jump()

func _start_charge() -> void:
	charging = true
	charge_start_time = Time.get_ticks_msec() / 1000.0
	charge_particles.emitting = true
	charge_particle_timer.start()
	accumulation_player.play()

func _do_jump() -> void:
	var next_platform: Node3D = get_parent().next_platform
	if next_platform == null:
		return
	
	var charge_time := Time.get_ticks_msec() / 1000.0 - charge_start_time
	charging = false
	charge_particles.emitting = false
	charge_particle_timer.stop()
	accumulation_player.stop()
	
	var player_pos := transform.origin
	var main_node: Node3D = get_parent()
	var current_platform: Node3D = main_node.current_platform
	var current_pos := current_platform.transform.origin
	var next_pos := next_platform.transform.origin
	
	# Calculate landing position
	var landing_pos: Vector3
	if abs(next_pos.x - current_pos.x) < 0.1:
		landing_pos = Vector3(player_pos.x, INITIAL_POS.y, player_pos.z - JUMP_SPEED * charge_time)
	else:
		landing_pos = Vector3(player_pos.x + JUMP_SPEED * charge_time, INITIAL_POS.y, player_pos.z)
	
	jump_active = true
	jump_start_pos = player_pos
	jump_end_pos = landing_pos
	jump_duration = max(charge_time / 2.0, 0.5)
	
	# Landing check
	var landed_next := _is_landed_on(next_platform, landing_pos)
	var landed_current := _is_landed_on(current_platform, landing_pos)
	
	if landed_next or landed_current:
		jump_falled = false
		if landed_next:
			GameState.add_score()
			GameState.score_up.emit(landing_pos + Vector3.UP * 0.5)
			main_node._on_player_landed()
	else:
		jump_falled = true
		if _is_touched(current_platform, landing_pos, 0.2):
			var fd := Vector3.LEFT if abs(landing_pos.x - player_pos.x) < 0.1 else Vector3.FORWARD
			_start_tilt_fall(landing_pos, fd)
		elif _is_touched(next_platform, landing_pos, 0.2):
			var fd: Vector3
			if abs(landing_pos.x - player_pos.x) < 0.1:
				fd = Vector3.LEFT if landing_pos.z < next_pos.z else Vector3.RIGHT
			else:
				fd = Vector3.BACK if landing_pos.x < next_pos.x else Vector3.FORWARD
			_start_tilt_fall(landing_pos, fd)
		else:
			_start_straight_fall(landing_pos)

func _animate_charge(delta: float) -> void:
	scale.x = minf(scale.x + CHARGE_XZ_RATE * delta, MAX_SCALE_XZ)
	scale.y = maxf(scale.y - CHARGE_Y_RATE * delta, MIN_SCALE_Y)
	scale.z = minf(scale.z + CHARGE_XZ_RATE * delta, MAX_SCALE_XZ)
	transform.origin.y = INITIAL_POS.y + (scale.y - 1.0) * 0.25

func _animate_jump(delta: float) -> void:
	var around_point := (jump_start_pos + jump_end_pos) / 2.0
	var rotate_axis: Vector3
	if abs(jump_end_pos.x - jump_start_pos.x) < 0.1:
		rotate_axis = Vector3.RIGHT
	else:
		rotate_axis = Vector3.BACK
	
	var angle := -(1.0 / jump_duration) * PI * delta
	var quat := Quaternion(rotate_axis, angle)
	var offset := transform.origin - around_point
	var new_pos := around_point + quat * offset
	
	if new_pos.y < INITIAL_POS.y:
		transform.origin = jump_end_pos
		transform.basis = Basis.IDENTITY
		scale = Vector3.ONE
		jump_active = false
		if not jump_falled:
			success_player.play()
	else:
		transform.origin = new_pos
		# Self-rotation (flip)
		var self_angle := -(1.0 / jump_duration) * TAU * delta
		rotate_object_local(rotate_axis, self_angle)

func _animate_fall(delta: float) -> void:
	match fall_type:
		FallType.STRAIGHT:
			if not fall_played_sound:
				fall_player.play()
				fall_played_sound = true
			if transform.origin.y < 0.5:
				fall_active = false
				GameState.change_state(GameState.State.GAME_OVER)
			else:
				transform.origin.y -= FALL_SPEED * delta
		FallType.TILT:
			if not fall_played_sound:
				fall_player.play()
				fall_played_sound = true
			if not fall_tilt_done:
				var pivot := Vector3(fall_pos.x, INITIAL_POS.y - 0.5, fall_pos.z)
				if transform.origin.y < pivot.y:
					fall_tilt_done = true
				else:
					var quat := Quaternion(fall_tilt_dir, PI / 2.0 * delta)
					var off := transform.origin - pivot
					transform.origin = pivot + quat * off
					transform.basis = Basis(quat) * transform.basis
			else:
				if transform.origin.y < 0.2:
					fall_active = false
					GameState.change_state(GameState.State.GAME_OVER)
				else:
					transform.origin.y -= FALL_SPEED * delta

func _start_straight_fall(pos: Vector3) -> void:
	fall_active = true
	fall_type = FallType.STRAIGHT
	fall_pos = pos
	fall_played_sound = false

func _start_tilt_fall(pos: Vector3, direction: Vector3) -> void:
	fall_active = true
	fall_type = FallType.TILT
	fall_pos = pos
	fall_tilt_dir = direction
	fall_tilt_done = false
	fall_played_sound = false

static func _is_landed_on(platform: Node3D, pos: Vector3) -> bool:
	if platform == null:
		return false
	var ppos := platform.transform.origin
	if platform.has_meta("shape"):
		var shape: String = platform.get_meta("shape")
		match shape:
			"cylinder":
				return abs(pos.x - ppos.x) < 0.75 and abs(pos.z - ppos.z) < 0.75
	return abs(pos.x - ppos.x) < 0.75 and abs(pos.z - ppos.z) < 0.75

static func _is_touched(platform: Node3D, pos: Vector3, radius: float) -> bool:
	if platform == null:
		return false
	var ppos := platform.transform.origin
	if platform.has_meta("shape"):
		var shape: String = platform.get_meta("shape")
		match shape:
			"cylinder":
				return abs(pos.x - ppos.x) < (0.75 + radius) and abs(pos.z - ppos.z) < (0.75 + radius)
	return abs(pos.x - ppos.x) < (0.75 + radius) and abs(pos.z - ppos.z) < (0.75 + radius)
```

Create `res://scripts/camera.gd`:
```gdscript
extends Camera3D

const INITIAL_POS := Vector3(-5.0, 8.0, 5.0)

var camera_step := Vector3.ZERO
var tracked_player_pos := Vector3(0.0, 1.5, 0.0)

func _ready() -> void:
	transform.origin = INITIAL_POS
	var target := Vector3.ZERO
	target.y = 1.0
	look_at(target, Vector3.UP)

func _process(_delta: float) -> void:
	if GameState.current_state != GameState.State.PLAYING:
		return
	
	var main_node: Node3D = get_parent()
	if main_node == null or main_node.player == null:
		return
	
	var player_pos := main_node.player.transform.origin
	var player := main_node.player
	
	# Only follow when player is idle (not jumping, not falling)
	if player.jump_active or player.fall_active:
		return
	
	var camera_destination := INITIAL_POS + player_pos
	
	if tracked_player_pos.distance_to(player_pos) > 0.1:
		var delta := camera_destination - transform.origin
		camera_step = 0.05 * delta
		tracked_player_pos = player_pos
	
	if transform.origin.distance_to(camera_destination) > Vector3.ZERO.distance_to(camera_step):
		transform.origin += camera_step
```

- [ ] **Step 2: Write main.gd**

Create `res://scripts/main.gd`:
```gdscript
extends Node3D

var player_scene: PackedScene = preload("res://scenes/player.tscn")
var platform_scene: PackedScene = preload("res://scenes/platform.tscn")

var player: Node3D = null
var current_platform: Node3D = null
var next_platform: Node3D = null

@onready var platforms_container: Node3D = $Platforms
@onready var start_sound: AudioStreamPlayer = $StartSound
@onready var prepare_timer: Timer = $PrepareTimer

func _ready() -> void:
	GameState.state_changed.connect(_on_state_changed)
	_enter_main_menu()

func _process(delta: float) -> void:
	# Animate current platform sinking/spring-back
	if GameState.current_state == GameState.State.PLAYING and current_platform:
		if player and player.charging:
			current_platform.scale.y = maxf(current_platform.scale.y - 0.15 * delta, 0.6)
		elif current_platform.scale.y < 1.0:
			current_platform.scale.y = lerpf(current_platform.scale.y, 1.0, 5.0 * delta)
			if current_platform.scale.y > 0.99:
				current_platform.scale.y = 1.0

func _on_state_changed(new_state: GameState.State) -> void:
	match new_state:
		GameState.State.MAIN_MENU:
			_enter_main_menu()
		GameState.State.PLAYING:
			_enter_playing()
		GameState.State.GAME_OVER:
			_enter_game_over()

func _enter_main_menu() -> void:
	_clear_player()
	_clear_platforms()
	$UI/MainMenu.visible = true
	$UI/GameOver.visible = false
	$UI/Scoreboard.visible = false

func _enter_playing() -> void:
	_clear_player()
	_clear_platforms()
	$UI/MainMenu.visible = false
	$UI/GameOver.visible = false
	$UI/Scoreboard.visible = true
	
	GameState.reset_score()
	
	# Spawn first platform
	current_platform = _spawn_platform(Vector3(0.0, 0.5, 0.0), true)
	
	# Spawn player
	player = player_scene.instantiate()
	add_child(player)
	player.transform.origin = Vector3(0.0, 1.5, 0.0)
	
	# Generate first next platform
	_generate_next_platform()
	
	# Start delay timer
	prepare_timer.start()
	start_sound.play()

func _enter_game_over() -> void:
	$UI/GameOver.visible = true
	$UI/Scoreboard.visible = false

func _clear_player() -> void:
	if player:
		player.queue_free()
		player = null

func _clear_platforms() -> void:
	for child in platforms_container.get_children():
		child.queue_free()
	current_platform = null
	next_platform = null

func _spawn_platform(pos: Vector3, is_current: bool) -> Node3D:
	var p := platform_scene.instantiate()
	p.transform.origin = pos
	p.is_current = is_current
	platforms_container.add_child(p)
	return p

func _generate_next_platform() -> void:
	if current_platform == null:
		return
	
	var current_pos := current_platform.transform.origin
	var rand_distance := randf() * 1.5 + 2.5  # 2.5 ~ 4.0
	
	var next_pos: Vector3
	if randi() % 2 == 0:
		next_pos = Vector3(current_pos.x + rand_distance, 0.5, current_pos.z)
	else:
		next_pos = Vector3(current_pos.x, 0.5, current_pos.z - rand_distance)
	
	next_platform = _spawn_platform(next_pos, false)

func _on_player_landed() -> void:
	if current_platform:
		current_platform.is_current = false
	current_platform = next_platform
	current_platform.is_current = true
	next_platform = null
	_generate_next_platform()
```

- [ ] **Step 3: Write scoreboard.gd**

Create `res://scripts/ui/scoreboard.gd`:
```gdscript
extends Control

var score_label: Label = null
var score_up_labels: Array[Label] = []

func _ready() -> void:
	GameState.score_changed.connect(_on_score_changed)
	GameState.score_up.connect(_on_score_up)
	
	# Create score label
	score_label = Label.new()
	score_label.add_theme_font_size_override("font_size", 40)
	score_label.add_theme_color_override("font_color", Color(0.5, 0.5, 1.0))
	score_label.text = "0"
	score_label.position = Vector2(30, 30)
	add_child(score_label)

func _on_score_changed(new_score: int) -> void:
	score_label.text = str(new_score)

func _on_score_up(landing_pos: Vector3) -> void:
	var label := Label.new()
	label.add_theme_font_size_override("font_size", 40)
	label.add_theme_color_override("font_color", Color(0.5, 0.5, 1.0))
	label.text = "+1"
	
	# Convert world pos to screen pos
	var camera: Camera3D = get_viewport().get_camera_3d()
	var screen_pos := camera.unproject_position(landing_pos)
	label.position = screen_pos
	add_child(label)
	score_up_labels.append(label)

func _process(delta: float) -> void:
	# Animate score-up labels: rise and fade
	if GameState.current_state != GameState.State.PLAYING:
		# Clean up all floating labels
		for label in score_up_labels:
			label.queue_free()
		score_up_labels.clear()
		return
	
	var to_remove: Array[int] = []
	for i in range(score_up_labels.size()):
		var label := score_up_labels[i]
		label.position.y -= 1.0 * delta * 100  # pixels per second
		var color := label.get_theme_color("font_color")
		color.a *= 0.97
		label.add_theme_color_override("font_color", color)
		if label.position.y < -40:
			to_remove.append(i)
	
	# Remove in reverse order
	to_remove.reverse()
	for i in to_remove:
		score_up_labels[i].queue_free()
		score_up_labels.remove_at(i)
```
